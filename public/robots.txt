# One Window Bayanihan — Department of Migrant Workers, Region VII

# Crawling is deliberately ALLOWED for public pages even before launch. Whether
# pages are INDEXED is controlled by SEARCH_INDEXING_ENABLED, which drives both
# the X-Robots-Tag header and the robots meta tag.
#
# Do not "Disallow: /" here to keep the site private: robots.txt also governs the
# link-preview crawlers used by Messenger, Viber, WhatsApp, Slack, LinkedIn and
# Twitter/X. Blocking them removes the preview card entirely, while doing nothing
# to prevent indexing — which the noindex directive handles properly.

User-agent: *
